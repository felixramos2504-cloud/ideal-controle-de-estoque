  function normalizeLookupText(value){ return String(value??'').toLocaleLowerCase('pt-BR').normalize('NFD').replace(/[\u0300-\u036f]/g,'').replace(/^[a-z]{2}:/,'').replace(/[-_]/g,' ').replace(/\s+/g,' ').trim(); }

  function externalProductName(product){
    const name=String(product?.product_name_pt||product?.product_name||product?.abbreviated_product_name||product?.generic_name_pt||product?.generic_name||'').trim();
    if(!name)return '';
    const brands=String(product?.brands||'').split(',').map(x=>x.trim()).filter(Boolean);
    const brand=brands[0]||'';
    return brand && !normalizeLookupText(name).includes(normalizeLookupText(brand)) ? brand+' '+name : name;
  }

    }
    return [...new Set(out.map(normalizeLookupText).filter(Boolean))];
  }

  function chooseSectionForProduct(product, productName=''){
    if(!state.sections.length)return null;
    const hints=categoryHints(product||{});
    const normalizedName=normalizeLookupText(productName||externalProductName(product||{}));
    const ranked=state.sections.map(section=>{
      const sectionNorm=normalizeLookupText(section.name);
      let score=0;
      for(const hint of hints){
        if(hint===sectionNorm) score=Math.max(score,100);
        else if(sectionNorm.length>3 && hint.includes(sectionNorm)) score=Math.max(score,50);
        else if(sectionNorm.length>3 && sectionNorm.includes(hint) && hint.length>4) score=Math.max(score,35);
      }
      const group=sectionCategoryGroups.find(g=>g.names.some(name=>{const n=normalizeLookupText(name);return sectionNorm===n || sectionNorm.includes(n) || (n.length>5 && n.includes(sectionNorm));}));
      if(group){
        for(const alias of group.aliases){
          const a=normalizeLookupText(alias);
          for(const hint of hints){
            if(hint===a) score=Math.max(score,85);
            else if(hint.includes(a) && a.length>3) score=Math.max(score,65);
            else if(a.includes(hint) && hint.length>5) score=Math.max(score,32);
          }
          if(normalizedName && a.length>3 && normalizedName.includes(a)) score=Math.max(score,12);
        }
      }
      return {section,score};
    }).sort((a,b)=>b.score-a.score);
    if(ranked[0]?.score>0)return ranked[0].section;
    const mercearia=state.sections.find(s=>normalizeLookupText(s.name)==='mercearia');
    if(mercearia)return mercearia;
    if(state.sections.length===1)return state.sections[0];
    return null;
  }
  async function fetchOpenFoodFactsProduct(ean){
    const fields='code,product_name,product_name_pt,abbreviated_product_name,generic_name,generic_name_pt,brands,quantity,product_quantity,product_quantity_unit,categories,categories_tags,categories_hierarchy,categories_tags_pt,categories_tags_en,main_category,main_category_pt,main_category_en,food_groups,food_groups_tags,pnns_groups_1,pnns_groups_2,pnns_groups_1_tags,pnns_groups_2_tags,product_type';
    const endpoints=[
      'https://world.openfoodfacts.org/api/v3/product/'+encodeURIComponent(ean)+'?product_type=all&cc=br&lc=pt&tags_lc=pt&fields='+encodeURIComponent(fields),
      'https://world.openfoodfacts.org/api/v2/product/'+encodeURIComponent(ean)+'.json?fields='+encodeURIComponent(fields)
    ];
    for(const url of endpoints){
      try{
        const controller=new AbortController(); const timeout=setTimeout(()=>controller.abort(),7000);
        let response;
        try{response=await fetch(url,{method:'GET',headers:{Accept:'application/json'},cache:'no-store',signal:controller.signal});}
        finally{clearTimeout(timeout);}
        if(!response.ok)continue;
        const data=await response.json();
        const product=data?.product;
        if(product && (externalProductName(product) || categoryHints(product).length))return product;
      }catch(_){ /* tenta o endpoint de compatibilidade; se ambos falharem, permite preenchimento manual */ }
    }
    return null;
  }
  async function lookupProductByEan(rawEan){
    const ean=normalizeBarcode(rawEan);
    if(!ean)return;
    $('pEan').value=ean;
    const token=++state.eanLookupToken;
    if(!/^\d{8}$|^\d{13}$/.test(ean)){
      setEanLookupStatus('Informe um EAN-8 ou EAN-13 válido para consultar a base de produtos.','warning');
      return;
    }
    const local=state.products.find(p=>p.ean===ean);
    if(local){
      $('productId').value=local.id;
      $('pName').value=local.name||'';
      $('pSection').value=local.sectionId||'';
      $('pUnit').value=local.unit||'UN';
      $('pStock').value=Number(local.stock||0);
      setEanLookupStatus('Este EAN já está cadastrado. Os dados locais foram carregados para edição.','success');
      return;
    }
    if(!eanChecksum(ean)){
      setEanLookupStatus('O dígito verificador deste EAN não confere. Confira os números antes de salvar.','warning');
    }else{
      setEanLookupStatus('Consultando a base pública Open Food Facts…');
    }
    let product;
    try{
      if(Object.prototype.hasOwnProperty.call(state.eanLookupCache,ean)) product=state.eanLookupCache[ean];
      else {product=await fetchOpenFoodFactsProduct(ean);state.eanLookupCache[ean]=product||null;}
    }catch(_){product=null;}
    if(token!==state.eanLookupToken)return;
    if(!product){
      const fallbackSection=chooseSectionForProduct(null,'');
      if(fallbackSection && !$('pSection').value)$('pSection').value=fallbackSection.id;
      setEanLookupStatus('EAN não localizado na base pública. O código foi mantido; preencha o nome e confira a seção antes de salvar.','warning');
      return;
    }
    const name=externalProductName(product);
    if(name)$('pName').value=name;
    const section=chooseSectionForProduct(product,name);
    if(section)$('pSection').value=section.id;
    const category=String(product.main_category_pt||product.main_category||product.categories||'').split(',')[0].trim();
    if(name && section){
      setEanLookupStatus('Produto encontrado na Open Food Facts: '+name+'. Seção sugerida: '+section.name+(category?' · Categoria na base: '+category:'')+'. Confira os dados antes de salvar.','success');
    }else if(name){
      setEanLookupStatus('Nome encontrado: '+name+'. Não consegui relacionar a categoria a uma seção cadastrada; selecione a seção manualmente antes de salvar.','warning');
    }else if(section){
      setEanLookupStatus('Produto localizado, mas a base não possui nome preenchido. Seção sugerida: '+section.name+'; complete o nome manualmente.','warning');
    }else{
      setEanLookupStatus('Produto encontrado, mas a base não informa nome ou uma categoria compatível. Complete o cadastro manualmente.','warning');
    }
  }
  function lookupProductFromEanField(){
    const raw=$('pEan').value.trim();
    if(!raw)return;
    state.eanLookupToken++; // invalida uma consulta anterior se o EAN foi alterado
    const ean=normalizeBarcode(raw);
    $('pEan').value=ean;
    if(!validEan(ean)){
      setEanLookupStatus('Digite ou leia um EAN-8 ou EAN-13. Códigos UPC-A de 12 dígitos são convertidos para EAN-13.','warning');
      return;
    }
    if(!eanChecksum(ean)){
      setEanLookupStatus('O código tem 8 ou 13 dígitos, mas o dígito verificador não confere. Confira a leitura.','warning');
      return;
    }
    lookupProductByEan(ean);
  }

  function eanChecksum(v){
    if(!validEan(v)) return false;
    const digits=v.split('').map(Number), check=digits.pop();
    let sum=0; for(let i=digits.length-1,weight=3;i>=0;i--,weight=weight===3?1:3) sum+=digits[i]*weight;
    return (10-(sum%10))%10===check;
  }
  async function startScanner(key, elementId, onScan){
    if(!window.Html5Qrcode){ message('Biblioteca de leitura não carregou. Verifique a conexão e tente novamente.','error'); return; }
    await stopScanner(key);
    const wrap = key==='product'?'productScannerWrap':key==='count'?'countScannerWrap':'inventoryScannerWrap';
    $(wrap).classList.remove('hidden');
    document.body.classList.add('scanner-open');
    state.scannerCaptured[key]=false;
    try{
      const scanner=new Html5Qrcode(elementId); state.scanners[key]=scanner;
      await scanner.start(
        {facingMode:'environment'},
        {
          fps:12,
          qrbox:(viewfinderWidth,viewfinderHeight)=>{
            const width=Math.max(220,Math.min(Math.floor(viewfinderWidth*0.9),700));
            const height=Math.max(90,Math.min(Math.floor(viewfinderHeight*0.28),220));
            return {width,height};
          },
          aspectRatio: 1.777,
          disableFlip:false,
          formatsToSupport:[Html5QrcodeSupportedFormats.EAN_13,Html5QrcodeSupportedFormats.EAN_8,Html5QrcodeSupportedFormats.UPC_A,Html5QrcodeSupportedFormats.UPC_E]
        },
        async decoded=>{
          const value=String(decoded||'').replace(/\D/g,'');
          if(!value || state.scannerCaptured[key]) return;
          state.scannerCaptured[key]=true;
          try{
            // Fecha a visualização primeiro para evitar leituras duplicadas e liberar a câmera.
            await stopScanner(key);
            onScan(value);
            if('vibrate' in navigator) navigator.vibrate(45);
          }catch(scanError){
            message('O código foi lido, mas não foi possível preencher o campo. '+(scanError.message||''),'error');
          }
        },
        ()=>{}
      );
    }catch(e){
      await stopScanner(key);
      message('Não foi possível abrir a câmera. Autorize o acesso e use o site por HTTPS. '+(e.message||''),'error');
    }
  }
  async function stopScanner(key){

    const wrap=key==='product'?'productScannerWrap':key==='count'?'countScannerWrap':'inventoryScannerWrap';

    state.scannerCaptured[key]=false;
    if(!Object.keys(state.scanners).length) document.body.classList.remove('scanner-open');
  }

  function historyMillis(value){
    const d=value&&typeof value.toDate==='function'?value.toDate():(value instanceof Date?value:new Date(value||0));
    return Number.isFinite(d.getTime())?d.getTime():0;
  }
  function historyDateKey(value){
    const ms=historyMillis(value);
    if(!ms)return '';
    const d=new Date(ms);
    return String(d.getFullYear())+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');
  }
  function historyDateInRange(value){
    const date=historyDateKey(value);
    const from=$('historyDateFrom').value;
    const to=$('historyDateTo').value;
    if(!date)return !from&&!to;
    if(from&&date<from)return false;
    if(to&&date>to)return false;
    return true;
  }

    const sectionId=$('historySection').value;
    const origin=$('historyOrigin').value;
    const result=$('historyResult').value;
    return state.counts.filter(item=>{
      const isInventory=!!item.inventoryId;
      if(origin==='inventory'&&!isInventory)return false;
      if(origin==='quick'&&isInventory)return false;
      if(sectionId&&historyCountSectionId(item)!==sectionId)return false;
      if(!historyDateInRange(item.createdAt))return false;
      const name=String(item.productName||'').toLocaleLowerCase('pt-BR');
      const ean=String(item.ean||'');
      const inventoryName=historyInventoryName(item.inventoryId).toLocaleLowerCase('pt-BR');
      if(q&&!name.includes(q)&&!ean.includes(q)&&!inventoryName.includes(q))return false;
      const diff=Number(item.countedQty||0)-Number(item.systemStock||0);
      if(result==='difference'&&diff===0)return false;
      if(result==='match'&&diff!==0)return false;
      return true;
    }).sort((a,b)=>historyMillis(b.createdAt)-historyMillis(a.createdAt));
  }

  function calculateInventoryCoverage(){
    if(!state.productsLoaded)return {ready:false,total:0,counted:0,missing:[],percent:0};
    const sectionId=state.inventory?state.inventory.sectionId:($('inventorySection')?$('inventorySection').value:null);
    const catalog=state.products.filter(p=>!sectionId||p.sectionId===sectionId);
    const countedIds=new Set(state.inventoryItems.map(it=>it.productId));
    const counted=catalog.filter(p=>countedIds.has(p.id)).length;
    const missing=catalog.filter(p=>!countedIds.has(p.id)).sort((a,b)=>String(a.name||'').localeCompare(String(b.name||''),'pt-BR'));
    return {ready:true,total:catalog.length,counted:counted,missing:missing,percent:catalog.length?Math.round(counted/catalog.length*100):0};
  }
  function renderInventoryCoverage(){
    const badge=$('inventoryCoverageBadge'),progress=$('inventoryCoverageProgress'),summary=$('inventoryCoverageSummary'),list=$('inventoryMissingList');
    if(!badge||!progress||!summary||!list)return;
    if(!state.inventory){
      badge.textContent='Aguardando inventário';progress.value=0;
      summary.textContent=state.productsLoaded?'Catálogo: '+state.products.length+' produto(s). Inicie um inventário para medir a cobertura da conferência.':'Carregando catálogo de produtos.';

  function inventoryDraftKey(user=state.user){return user?.uid?'ideal-estoque-inventory-draft-v1:'+user.uid:null;}
  function setInventoryDraftStatus(text){const el=$('inventoryDraftStatus');if(el)el.textContent=text;}
  function persistInventoryDraft(){
    const key=inventoryDraftKey();
    if(!key)return;
    try{
      if(!state.inventory){
        localStorage.removeItem(key);
        setInventoryDraftStatus('Nenhum inventário salvo neste dispositivo.');
        return;
      }
      const payload={
        version:1,
        savedAt:new Date().toISOString(),
        inventory:{
          name:String(state.inventory.name||'Inventário sem título'),
          sectionId:state.inventory.sectionId||null,
          startedAt:state.inventory.startedAt||new Date().toISOString()
        },
        items:state.inventoryItems.map(it=>({
          productId:String(it.productId||''),
          productName:String(it.productName||'Produto'),
          ean:String(it.ean||''),
          systemStock:Math.max(0,Number(it.systemStock)||0),
          countedQty:Math.max(0,Number(it.countedQty)||0),
          sectionId:it.sectionId||null,
          unit:String(it.unit||'UN')
        })).filter(it=>it.productId)
      };
      localStorage.setItem(key,JSON.stringify(payload));
      setInventoryDraftStatus('Rascunho salvo automaticamente neste dispositivo às '+new Date().toLocaleTimeString('pt-BR',{hour:'2-digit',minute:'2-digit'})+'.');
    }catch(e){
      setInventoryDraftStatus('Não foi possível salvar o rascunho automaticamente. Verifique o espaço disponível no navegador.');
    }
  }
  function restoreInventoryDraft(user){
    const key=inventoryDraftKey(user);
    state.inventory=null;
    state.inventoryItems=[];
    if(!key){setInventoryDraftStatus('Entre no sistema para recuperar o rascunho.');return;}
    try{
      const raw=localStorage.getItem(key);
      if(!raw){setInventoryDraftStatus('Nenhum inventário salvo neste dispositivo.');renderInventory();return;}
      const draft=JSON.parse(raw);
      if(draft?.version!==1||!draft.inventory||!Array.isArray(draft.items))throw new Error('Rascunho inválido.');
      state.inventory={
        name:String(draft.inventory.name||'Inventário sem título'),
        sectionId:draft.inventory.sectionId||null,
        startedAt:draft.inventory.startedAt||new Date().toISOString()
      };
      state.inventoryItems=draft.items.filter(it=>it&&it.productId).map(it=>({
        productId:String(it.productId),
        productName:String(it.productName||'Produto'),
        ean:String(it.ean||''),
        systemStock:Math.max(0,Number(it.systemStock)||0),
        countedQty:Math.max(0,Number(it.countedQty)||0),
        sectionId:it.sectionId||null,
        unit:String(it.unit||'UN')
      }));
      $('inventoryName').value=state.inventory.name;
      setInventoryDraftStatus('Rascunho recuperado automaticamente deste dispositivo. Confira os itens antes de finalizar.');
      renderInventory();
    }catch(e){
      try{localStorage.removeItem(key);}catch(_){}
      state.inventory=null;state.inventoryItems=[];
      setInventoryDraftStatus('O rascunho salvo estava inválido e foi descartado. Você pode iniciar um novo inventário.');
      renderInventory();
    }
  }
  function discardInventoryDraft(){
    if(!state.inventory)return;
    if(!window.confirm('Descartar o inventário em andamento e o rascunho salvo neste dispositivo? Esta ação não altera o estoque oficial.'))return;
    state.inventory=null;
    state.inventoryItems=[];
    $('inventoryName').value='';
    $('inventorySection').value='';
    persistInventoryDraft();
    renderInventory();
    message('Rascunho descartado. O estoque oficial não foi alterado.','success');
  }

  function plainBackupValue(value){
    if(value&&typeof value.toDate==='function')return value.toDate().toISOString();
    if(value instanceof Date)return value.toISOString();

    if(value&&typeof value==='object')return Object.fromEntries(Object.entries(value).map(([key,val])=>[key,plainBackupValue(val)]));
    return value;
  }

  function runIntegrityAnalysis(products,sections){
    const issues=[];
    const add=(severity,code,item,details)=>issues.push({severity:severity,code:code,item:item,details:details});
    const sectionIds=new Set(sections.map(section=>section.id));
    const eanGroups=new Map();
    products.forEach(p=>{
      const name=String(p.name||'').trim()||'(sem nome)';
      const ean=String(p.ean||'').trim();
      if(!String(p.name||'').trim())add('error','missing_name',name,'Produto sem nome. ID: '+p.id);
      if(!ean)add('warning','missing_ean',name,'Produto sem EAN. ID: '+p.id);
      else{
        if(!validEan(ean)||!eanChecksum(ean))add('error','invalid_ean',name,'EAN inválido: '+ean+'. ID: '+p.id);
        const normalized=ean.replace(/\s/g,'');
        if(!eanGroups.has(normalized))eanGroups.set(normalized,[]);
        eanGroups.get(normalized).push(p);
      }
      if(!p.sectionId||!sectionIds.has(p.sectionId))add('error','missing_section',name,'Seção não cadastrada ou removida. ID do produto: '+p.id+'; sectionId: '+(p.sectionId||'vazio'));
      const stock=Number(p.stock);
      if(!Number.isFinite(stock)||stock<0)add('error','invalid_stock',name,'Saldo inválido: '+String(p.stock)+'; ID: '+p.id);
    });
    eanGroups.forEach((group,ean)=>{
      if(group.length>1)add('error','duplicate_ean',ean,'EAN repetido em '+group.length+' produtos: '+group.map(p=>String(p.name||p.id)+' [ID '+p.id+']').join('; '));
    });
    const sectionNameGroups=new Map();
    sections.forEach(section=>{
      const key=normalizeLookupText(section.name||'');
      if(!key){add('error','empty_section_name','Seção sem nome','ID da seção: '+section.id);return;}
      if(!sectionNameGroups.has(key))sectionNameGroups.set(key,[]);
      sectionNameGroups.get(key).push(section);
    });
    sectionNameGroups.forEach((group,key)=>{
      if(group.length>1)add('warning','duplicate_section',key,'Nome de seção duplicado ou equivalente: '+group.map(s=>String(s.name)+' [ID '+s.id+']').join('; '));
    });
    const errors=issues.filter(issue=>issue.severity==='error').length;
    const warnings=issues.filter(issue=>issue.severity==='warning').length;
    return {issues:issues,summary:{products:products.length,sections:sections.length,errors:errors,warnings:warnings,eanDuplicates:issues.filter(i=>i.code==='duplicate_ean').length},checkedAt:new Date()};
  }

  function getAnalysisPeriodCounts(){
    const period=$('analysisPeriod').value;

  function analyzeCounts(rows){
    const byDay=new Map(),byProduct=new Map(),byUser=new Map();
    let different=0;
    rows.forEach(item=>{
      const stock=Number(item.systemStock||0),counted=Number(item.countedQty||0),diff=counted-stock;
      const hasDiff=diff!==0;
      if(hasDiff)different++;
      const day=historyDateKey(item.createdAt)||'Sem data';
      if(!byDay.has(day))byDay.set(day,{day,total:0,different:0});
      const dayRow=byDay.get(day);dayRow.total++;if(hasDiff)dayRow.different++;
      if(hasDiff){
        const key=String(item.productId||item.ean||item.productName||'produto');
        if(!byProduct.has(key))byProduct.set(key,{key,productName:item.productName||'Produto',ean:item.ean||'',unit:item.unit||'UN',occurrences:0,absoluteDifference:0,maxDifference:0});
        const prod=byProduct.get(key);prod.occurrences++;prod.absoluteDifference+=Math.abs(diff);prod.maxDifference=Math.max(prod.maxDifference,Math.abs(diff));if(item.unit)prod.unit=item.unit;
      }
      const userKey=String(item.userId||item.userEmail||'unknown');
      if(!byUser.has(userKey))byUser.set(userKey,{userId:userKey,email:item.userEmail||'Responsável não identificado',total:0,different:0});
      const user=byUser.get(userKey);user.total++;if(hasDiff)user.different++;
    });
    return {
      total:rows.length,
      different:different,
      matching:rows.length-different,
      accuracy:rows.length?Math.round((rows.length-different)/rows.length*100):0,
      uniqueProducts:new Set(rows.map(item=>String(item.productId||item.ean||item.productName||''))).size,
      products:[...byProduct.values()].sort((a,b)=>b.occurrences-a.occurrences||b.absoluteDifference-a.absoluteDifference||a.productName.localeCompare(b.productName,'pt-BR')),
      users:[...byUser.values()].map(u=>({...u,matching:u.total-u.different,accuracy:u.total?Math.round((u.total-u.different)/u.total*100):0})).sort((a,b)=>b.total-a.total||a.email.localeCompare(b.email,'pt-BR')),
      days:[...byDay.values()].sort((a,b)=>a.day.localeCompare(b.day)).slice(-14)
    };
  }

  function stockCsvDelimiter(headerLine){
    const semicolon=(headerLine.match(/;/g)||[]).length;
    const comma=(headerLine.match(/,/g)||[]).length;
    return semicolon>=comma?';':',';
  }
  function parseStockCsvLine(line,delimiter){
    const cells=[];let field='',quoted=false;
    for(let i=0;i<line.length;i++){
      const ch=line[i];
      if(ch==='"'){
        if(quoted&&line[i+1]==='"'){field+='"';i++;}
        else quoted=!quoted;
      }else if(ch===delimiter&&!quoted){cells.push(field.trim());field='';}
      else field+=ch;
    }
    cells.push(field.trim());return cells;
  }

  async function prepareStockImport(){
    if(bulkStockImportBusy)return;
    try{
      await requireRole('gerente');
      const file=$('bulkStockFile').files?.[0];
      if(!file)throw new Error('Selecione primeiro o arquivo CSV.');
      if(file.size>10*1024*1024)throw new Error('O arquivo ultrapassa 10 MB. Divida a planilha em arquivos menores.');
      const content=(await file.text()).replace(/^\uFEFF/,'').replace(/\r\n?/g,'\n');
      const nonEmpty=content.split('\n').filter(line=>line.trim());
      if(nonEmpty.length<2)throw new Error('A planilha precisa ter o cabeçalho e pelo menos uma linha de produto.');
      const delimiter=stockCsvDelimiter(nonEmpty[0]),records=nonEmpty.map(line=>parseStockCsvLine(line,delimiter));
      const headers=records.shift().map(stockImportHeader);
      const columns={
        ean:stockImportColumn(headers,['ean','codigo ean','codigo de barras','codigo barras','gtin','barcode']),
        name:stockImportColumn(headers,['produto','nome','nome do produto','descricao','descricao do produto','product']),
        section:stockImportColumn(headers,['secao','categoria','departamento','section']),
        unit:stockImportColumn(headers,['unidade','unidade de medida','unit']),
        stock:stockImportColumn(headers,['saldo atual','saldo','estoque atual','quantidade atual','quantidade','estoque','stock','quantity'])
      };
      const missing=[];
      if(columns.ean<0)missing.push('EAN');
      if(columns.name<0)missing.push('Produto');
      if(columns.section<0)missing.push('Seção');
      if(columns.stock<0)missing.push('Saldo atual');
      if(missing.length)throw new Error('Faltam colunas obrigatórias: '+missing.join(', ')+'. Baixe o modelo CSV para ver o formato.');
      if(records.length>10000)throw new Error('A planilha tem mais de 10.000 linhas. Divida em arquivos menores.');
      const sectionMap=new Map(state.sections.map(section=>[normalizeLookupText(section.name),section]));
      const productsByEan=new Map();
      const eanDuplicatesInCatalog=new Set();
      state.products.filter(p=>p.ean).forEach(p=>{
        const ean=normalizeBarcode(p.ean);
        if(productsByEan.has(ean))eanDuplicatesInCatalog.add(ean);
        else productsByEan.set(ean,p);
      });
      const parsed=records.map((record,index)=>{
        const value=col=>col>=0?String(record[col]??'').trim():'';
        const ean=normalizeBarcode(value(columns.ean).replace(/\D/g,''));
        const name=value(columns.name),sectionText=value(columns.section),section=sectionMap.get(normalizeLookupText(sectionText));
        const unitRaw=value(columns.unit)||'UN',unit=stockImportUnit(unitRaw),quantity=parseStockImportQuantity(value(columns.stock)),errors=[];

        if(!section)errors.push('Seção não cadastrada: '+(sectionText||'vazia'));
        if(!unit)errors.push('Unidade inválida: '+unitRaw);
        if(!Number.isFinite(quantity)||quantity<=0)errors.push('Saldo deve ser numérico e maior que zero');
        if(eanDuplicatesInCatalog.has(ean))errors.push('EAN duplicado no cadastro atual; corrija o cadastro antes de importar');
        const existing=productsByEan.get(ean);
        return {line:index+2,ean,name,sectionText,sectionId:section?.id||'',unit,quantity,existingId:existing?.id||'',existingName:existing?.name||'',valid:errors.length===0,errors};
      });
      const fileCounts=new Map();
      parsed.forEach(row=>{if(row.ean)fileCounts.set(row.ean,(fileCounts.get(row.ean)||0)+1);});
      parsed.forEach(row=>{if(row.ean&&fileCounts.get(row.ean)>1){row.valid=false;row.errors.push('EAN repetido nesta planilha');}});

      renderStockImportControls();
      message(invalidCount?'A planilha tem linhas inválidas. Corrija os erros indicados e confira novamente.':fmt(validCount)+' linha(s) prontas para importação. Revise a prévia antes de gravar.','success');
    }catch(err){
      bulkStockImportRows=[];bulkStockImportReady=false;renderStockImportControls();
      if($('bulkStockImportStatus'))$('bulkStockImportStatus').textContent=err.message||'Não foi possível ler a planilha.';

      if(bulkStockImportRows.some(row=>!row.valid))throw new Error('A importação foi bloqueada. Corrija todas as linhas inválidas e confira novamente.');
      if(!window.confirm('Confirma alimentar o estoque com '+bulkStockImportRows.length+' produto(s) do arquivo "'+bulkStockImportFileName+'"? Os itens não presentes no arquivo não serão excluídos.'))return;
      const chunkSize=400;
      for(let offset=0;offset<bulkStockImportRows.length;offset+=chunkSize){
        const batch=db.batch(),chunk=bulkStockImportRows.slice(offset,offset+chunkSize);
        chunk.forEach(row=>{
          const data={ean:row.ean,name:row.name,sectionId:row.sectionId,unit:row.unit,stock:row.quantity,updatedAt:firebase.firestore.FieldValue.serverTimestamp(),updatedBy:state.user.uid};
          if(row.existingId){batch.update(db.collection('products').doc(row.existingId),data);updatedCount++;}
          else{const ref=db.collection('products').doc();batch.set(ref,{...data,createdAt:firebase.firestore.FieldValue.serverTimestamp(),createdBy:state.user.uid});createdCount++;}
        });
        await batch.commit();processed+=chunk.length;
        $('bulkStockImportStatus').textContent='Gravando no Firestore: '+fmt(processed)+' de '+fmt(bulkStockImportRows.length)+' produto(s)...';
      }
      await recordAudit('stock_bulk_imported','products','bulk-import','Estoque alimentado por planilha: '+bulkStockImportFileName,{fileName:bulkStockImportFileName,createdCount,updatedCount,rowsProcessed:processed});
      bulkStockImportRows=[];bulkStockImportReady=false;bulkStockImportFileName='';$('bulkStockFile').value='';$('bulkStockImportSummary').innerHTML='';
      $('bulkStockImportTable').innerHTML='<tr><td colspan="7">Importação concluída. Os dados serão atualizados no painel em tempo real.</td></tr>';
      $('bulkStockImportFooter').textContent='Importação concluída. Confira o saldo atual no dashboard e no relatório.';
      $('bulkStockImportStatus').textContent='Concluído: '+fmt(createdCount)+' produto(s) cadastrado(s) e '+fmt(updatedCount)+' saldo(s) atualizado(s).';
      message('Estoque alimentado com sucesso: '+fmt(createdCount)+' produto(s) novo(s) e '+fmt(updatedCount)+' atualizado(s).','success');
      renderStockImportControls();renderDashboard();renderStockReport();renderProducts();
    }catch(err){
      message('A importação não foi concluída integralmente: '+(err.message||err)+'. Se já houver páginas gravadas, confira o estoque antes de tentar novamente.','error');
      if($('bulkStockImportStatus'))$('bulkStockImportStatus').textContent='Falha após '+fmt(processed)+' linha(s) gravada(s). '+(err.message||err);
    }finally{bulkStockImportBusy=false;renderStockImportControls();}
  }
  async function saveProduct(e){
    e.preventDefault();try{
      await requireRole('gerente');
      const id=$('productId').value,oldProduct=id?state.products.find(p=>p.id===id):null;
      const ean=$('pEan').value.trim().replace(/\s/g,'');
      if(ean&&(!validEan(ean)||!eanChecksum(ean)))throw new Error('EAN inválido. Confira o código de 8 ou 13 dígitos e o dígito verificador.');
      const data={ean,name:$('pName').value.trim(),sectionId:$('pSection').value,unit:$('pUnit').value,stock:Number($('pStock').value),updatedAt:firebase.firestore.FieldValue.serverTimestamp(),updatedBy:state.user.uid};
      if(!data.sectionId)throw new Error('Selecione uma seção.');
      if(state.products.some(p=>p.ean===ean&&ean&&p.id!==id))throw new Error('Este EAN já está cadastrado em outro produto.');
      if(!data.name||data.stock<0)throw new Error('Confira nome e quantidade de estoque.');
      let productId=id;
      if(id){await db.collection('products').doc(id).update(data);}
      else{data.createdAt=firebase.firestore.FieldValue.serverTimestamp();data.createdBy=state.user.uid;const ref=await db.collection('products').add(data);productId=ref.id;}
      await recordAudit(id?'product_updated':'product_created','product',productId,(id?'Produto atualizado: ':'Produto cadastrado: ')+data.name,{productName:data.name,ean:data.ean,sectionName:sectionName(data.sectionId),oldStock:oldProduct?Number(oldProduct.stock||0):null,newStock:Number(data.stock),unit:data.unit});
      $('productForm').reset();$('productId').value='';$('pStock').value='0';setEanLookupStatus('Produto salvo. Leia outro EAN para consultar automaticamente o próximo produto.','success');message('Produto salvo com sucesso.','success');
    }catch(err){message(err.message,'error');}
  }

  async function saveQuickCount(){
    const btn=$('saveCountBtn');
    if(btn.disabled)return;
    btn.disabled=true;
    try{
      if(!state.user)throw new Error('Entre no sistema.');
      const rawQty=$('countQty').value.trim();
      if(rawQty==='')throw new Error('Informe a quantidade física contada antes de registrar.');
      const qty=Number(rawQty);
      if(!Number.isFinite(qty)||qty<0)throw new Error('Quantidade inválida. Informe zero ou um número positivo.');
      const p=await lookupCountProduct();
      if(!p)throw new Error('Produto não encontrado. Verifique o EAN ou solicite o cadastro.');
      const systemStock=Number(p.stock||0);
      const difference=qty-systemStock;
      if(difference!==0){
        const sign=difference>0?'+':'';
        const ok=window.confirm('CONFIRME A CONTAGEM\n\nProduto: '+p.name+'\nEstoque cadastrado: '+fmt(systemStock)+' '+(p.unit||'UN')+'\nQuantidade física informada: '+fmt(qty)+' '+(p.unit||'UN')+'\nDiferença: '+sign+fmt(difference)+' '+(p.unit||'UN')+'\n\nA contagem rápida apenas registra a conferência; não altera o estoque oficial.\n\nDeseja registrar esses números?');
        if(!ok)return;
      }
      await db.collection('counts').add({productId:p.id,productName:p.name,ean:p.ean,sectionId:p.sectionId,unit:p.unit||'UN',systemStock:systemStock,countedQty:qty,difference:difference,userId:state.user.uid,userEmail:state.user.email||'',createdAt:firebase.firestore.FieldValue.serverTimestamp(),inventoryId:null});
      await recordAudit('quick_count_registered','count',p.id,'Contagem rápida registrada: '+p.name,{productName:p.name,ean:p.ean,sectionName:sectionName(p.sectionId),systemStock:systemStock,countedQty:qty,difference:difference,unit:p.unit||'UN'});
      $('countEan').value='';
      $('countQty').value='';
      $('countProductInfo').classList.add('hidden');
      message('Contagem registrada para conferência. O estoque oficial não foi alterado.','success');
    }catch(e){message(e.message,'error');}
    finally{btn.disabled=false;}
  }

  async function finishInventory(){try{
      await requireRole('gerente');
      if(!state.inventory||!state.inventoryItems.length)throw new Error('Não há inventário com produtos para finalizar.');
      if(!state.productsLoaded)throw new Error('Aguarde o carregamento do catálogo antes de finalizar.');
      const coverage=calculateInventoryCoverage();
      if(!coverage.total)throw new Error('Não há produtos no escopo do inventário. Confira a seção selecionada.');
      if(coverage.missing.length){
        const preview=coverage.missing.slice(0,5).map(p=>p.name||p.ean||'Produto').join(', ');
        const more=coverage.missing.length>5?' e mais '+(coverage.missing.length-5)+' produto(s)':'';
        const approved=window.confirm('ATENÇÃO: este inventário cobre '+coverage.counted+' de '+coverage.total+' produto(s). Ainda faltam '+coverage.missing.length+' produto(s) sem contagem.\n\nPendentes: '+preview+more+'.\n\nSe finalizar agora, somente os produtos contados terão seus saldos atualizados. Os produtos não contados NÃO serão zerados nem alterados.\n\nDeseja finalizar mesmo assim?');
        if(!approved)return;
      }
      const batch=db.batch(),invRef=db.collection('inventories').doc();
      batch.set(invRef,{name:state.inventory.name,sectionId:state.inventory.sectionId,startedAt:state.inventory.startedAt,finishedAt:firebase.firestore.FieldValue.serverTimestamp(),status:'finalizado',itemCount:state.inventoryItems.length,coverageTotal:coverage.total,coverageCounted:coverage.counted,coveragePercent:coverage.percent,userId:state.user.uid,userEmail:state.user.email||''});
      for(const it of state.inventoryItems){
        const pRef=db.collection('products').doc(it.productId);
        batch.update(pRef,{stock:Number(it.countedQty),updatedAt:firebase.firestore.FieldValue.serverTimestamp(),updatedBy:state.user.uid});
        const cRef=db.collection('counts').doc();
        batch.set(cRef,{...it,systemStock:Number(it.systemStock),countedQty:Number(it.countedQty),difference:Number(it.countedQty)-Number(it.systemStock),userId:state.user.uid,userEmail:state.user.email||'',createdAt:firebase.firestore.FieldValue.serverTimestamp(),inventoryId:invRef.id});
      }
      await batch.commit();
      await recordAudit('inventory_finalized','inventory',invRef.id,'Inventário finalizado: '+state.inventory.name,{inventoryName:state.inventory.name,sectionName:state.inventory.sectionId?sectionName(state.inventory.sectionId):'Todas as seções',itemCount:state.inventoryItems.length,coverageTotal:coverage.total,coverageCounted:coverage.counted,coveragePercent:coverage.percent});
      state.inventory=null;state.inventoryItems=[];persistInventoryDraft();renderInventory();message('Inventário finalizado. Estoques atualizados e diferenças registradas no histórico; o rascunho local foi removido.','success');
    }catch(e){message('Não foi possível finalizar o inventário: '+e.message,'error');}}
