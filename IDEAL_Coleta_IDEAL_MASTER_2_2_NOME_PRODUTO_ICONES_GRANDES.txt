    const brand=String(product?.brands||'').split(',').map(value=>value.trim()).filter(Boolean)[0]||'';
    return brand&&!normalizeText(name).includes(normalizeText(brand))?(brand+' '+name).slice(0,160):name.slice(0,160);
  }
  async function lookupPublicProductName(ean){
    if(Object.prototype.hasOwnProperty.call(productNameLookupCache,ean))return productNameLookupCache[ean];
    const fields='code,product_name,product_name_pt,abbreviated_product_name,generic_name,generic_name_pt,brands';
    const urls=[
      'https://world.openfoodfacts.org/api/v3/product/'+encodeURIComponent(ean)+'?product_type=all&cc=br&lc=pt&tags_lc=pt&fields='+encodeURIComponent(fields),
      'https://world.openfoodfacts.org/api/v2/product/'+encodeURIComponent(ean)+'.json?fields='+encodeURIComponent(fields),
      'https://world.openbeautyfacts.org/api/v2/product/'+encodeURIComponent(ean)+'.json?fields='+encodeURIComponent(fields),
      'https://world.openproductsfacts.org/api/v2/product/'+encodeURIComponent(ean)+'.json?fields='+encodeURIComponent(fields)
    ];
    const names=await Promise.all(urls.map(async url=>{
      try{
        const controller=new AbortController();
        const timeout=setTimeout(()=>controller.abort(),3500);
        let response;
        try{response=await fetch(url,{method:'GET',headers:{Accept:'application/json'},cache:'no-store',signal:controller.signal});}
        finally{clearTimeout(timeout);}
        if(!response.ok)return '';
        const data=await response.json();
        return getPublicProductName(data?.product);
      }catch(_){return '';}
    }));
    const name=names.find(value=>!!value)||null;
    productNameLookupCache[ean]=name;
    return name;
  }

      return;
    }
    const oldName=String(product?.name||'').trim();
    const needsPublicLookup=!oldName||/^Produto EAN\s/i.test(oldName);
    selectedProduct={id:product?product.id:'ean_'+ean,ean,name:needsPublicLookup?'':oldName,sectionId:product?.sectionId||'',unit:product?.unit||'UN',stock:Number(product?.stock||0),isNew:!product,isLookingUpName:needsPublicLookup,externalNameFound:false};

    $('quantityInput').focus();
    if(needsPublicLookup){
      const publicName=await lookupPublicProductName(ean);
      if(token!==productNameLookupToken||!selectedProduct||selectedProduct.ean!==ean)return;
      selectedProduct.isLookingUpName=false;selectedProduct.externalNameFound=!!publicName;selectedProduct.name=publicName||oldName||('Produto EAN '+ean);
      $('selectedProductName').textContent=selectedProduct.name;

    }
  }

  async function sendCollection(){
    if(busy||!collectionItems.length)return;
    if(!window.confirm('Enviar '+collectionItems.length+' produto(s) e substituir seus saldos pelas quantidades coletadas? Os demais produtos não serão modificados.'))return;
    busy=true;renderCollection();$('collectionStatus').textContent='Transmitindo '+fmt(collectionItems.length)+' produto(s) ao Firestore para atualizar o dashboard...';
    let sent=0;
    try{
      const freshProducts=new Map(products.map(product=>[product.id,product]));
      const freshByEan=new Map(products.filter(product=>product.ean).map(product=>[normalizeEan(product.ean),product]));
      for(let offset=0;offset<collectionItems.length;offset+=400){
        const batch=db.batch(),chunk=collectionItems.slice(offset,offset+400);
        chunk.forEach(item=>{
          const itemEan=normalizeEan(item.ean);
          const byEan=freshByEan.get(itemEan);
          const byId=freshProducts.get(item.id);
          const existing=byEan||(byId&&normalizeEan(byId.ean)===itemEan?byId:null);
          if(existing){
            const updateData={stock:Number(item.quantity),updatedAt:firebase.firestore.FieldValue.serverTimestamp(),updatedBy:auth.currentUser.uid};
            const existingName=String(existing.name||'');
            const collectedName=String(item.name||'').trim();
            if(item.externalNameFound&&/^Produto EAN\s/i.test(existingName)&&collectedName&&!/^Produto EAN\s/i.test(collectedName))updateData.name=collectedName.slice(0,160);
            batch.update(db.collection('products').doc(existing.id),updateData);
          }else{
            const newId='ean_'+itemEan,occupied=freshProducts.get(newId);
            if(occupied&&normalizeEan(occupied.ean)!==itemEan)throw new Error('Conflito no identificador do EAN '+item.ean+'. Nenhuma linha desse lote foi enviada; avise a gerência.');
            const ref=db.collection('products').doc(newId);
            const collectedName=String(item.name||'').trim();
            const productName=item.externalNameFound&&collectedName&&!/^Produto EAN\s/i.test(collectedName)?collectedName.slice(0,160):'Produto EAN '+itemEan;
            batch.set(ref,{ean:itemEan,name:productName,sectionId:'',unit:'UN',stock:Number(item.quantity),createdAt:firebase.firestore.FieldValue.serverTimestamp(),createdBy:auth.currentUser.uid,updatedAt:firebase.firestore.FieldValue.serverTimestamp(),updatedBy:auth.currentUser.uid});
          }
        });
        await batch.commit();sent+=chunk.length;$('collectionStatus').textContent='Enviados '+fmt(sent)+' de '+fmt(collectionItems.length)+' produto(s)...';
      }
      try{
        await db.collection('auditLogs').add({action:'stock_collection_sent',entityType:'products',entityId:'collection',summary:'Coleta de estoque enviada: '+sent+' produto(s)',details:{rowsProcessed:sent,items:collectionItems.map(item=>({productName:item.name,ean:item.ean,newStock:Number(item.quantity)}))},userId:auth.currentUser.uid,userEmail:String(auth.currentUser.email||''),role:currentRole,createdAt:firebase.firestore.FieldValue.serverTimestamp()});
      }catch(auditError){console.warn('Não foi possível registrar a auditoria da coleta:',auditError);}
      collectionItems=[];selectedProduct=null;$('selectedProductCard').classList.add('hidden');$('eanInput').value='';
      saveCollectionDraft();renderCollection();const transmittedAt=new Date().toLocaleString('pt-BR');$('collectionStatus').textContent='Transmissão concluída às '+transmittedAt+'. Os saldos foram gravados no Firestore; o dashboard recebe as alterações automaticamente em tempo real.';
      showMessage(fmt(sent)+' produto(s) transmitidos. O dashboard principal receberá os novos saldos em tempo real.','success');
    }catch(err){
      $('collectionStatus').textContent='Falha no envio. Confira as permissões e tente novamente.';
      const denied=String(err.code||'').includes('permission-denied')||/permission|permissão/i.test(err.message||'');
      showMessage(denied?'O Firebase bloqueou a gravação. Publique no Console do Firebase as regras atualizadas do arquivo firestore.rules, que autorizam a inclusão automática de EANs coletados. Erro: '+(err.message||'permissão negada'):'Não foi possível enviar toda a coleta: '+(err.message||err),'error');
    }finally{busy=false;renderCollection();}
  }
