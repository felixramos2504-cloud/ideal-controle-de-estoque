  function eanChecksum(v){

    const q=$('historySearch').value.trim().toLocaleLowerCase('pt-BR');
    const sectionId=$('historySection').value;
    const origin=$('historyOrigin').value;
    const result=$('historyResult').value;
    return state.counts.filter(item=>{
      const isInventory=!!item.inventoryId;
      if(origin==='inventory'&&!isInventory)return false;
      if(origin==='quick'&&isInventory)return false;
      if(sectionId&&historyCountSectionId(item)!==sectionId)return false;
      if(!historyDateInRange(item.createdAt))return false;
      const name=String(item.productName||'').toLocaleLowerCase('pt-BR');
      const ean=String(item.ean||'');
      const inventoryName=historyInventoryName(item.inventoryId).toLocaleLowerCase('pt-BR');
      if(q&&!name.includes(q)&&!ean.includes(q)&&!inventoryName.includes(q))return false;
      const diff=Number(item.countedQty||0)-Number(item.systemStock||0);
      if(result==='difference'&&diff===0)return false;
      if(result==='match'&&diff!==0)return false;
      return true;
    }).sort((a,b)=>historyMillis(b.createdAt)-historyMillis(a.createdAt));
  }

  function calculateInventoryCoverage(){
    if(!state.productsLoaded)return {ready:false,total:0,counted:0,missing:[],percent:0};
    const sectionId=state.inventory?state.inventory.sectionId:($('inventorySection')?$('inventorySection').value:null);
    const catalog=state.products.filter(p=>!sectionId||p.sectionId===sectionId);
    const countedIds=new Set(state.inventoryItems.map(it=>it.productId));
    const counted=catalog.filter(p=>countedIds.has(p.id)).length;
    const missing=catalog.filter(p=>!countedIds.has(p.id)).sort((a,b)=>String(a.name||'').localeCompare(String(b.name||''),'pt-BR'));
    return {ready:true,total:catalog.length,counted:counted,missing:missing,percent:catalog.length?Math.round(counted/catalog.length*100):0};
  }

  function persistInventoryDraft(){
    const key=inventoryDraftKey();
    if(!key)return;
    try{
      if(!state.inventory){
        localStorage.removeItem(key);
        setInventoryDraftStatus('Nenhum inventário salvo neste dispositivo.');
        return;
      }
      const payload={
        version:1,
        savedAt:new Date().toISOString(),
        inventory:{
          name:String(state.inventory.name||'Inventário sem título'),
          sectionId:state.inventory.sectionId||null,
          startedAt:state.inventory.startedAt||new Date().toISOString()
        },
        items:state.inventoryItems.map(it=>({
          productId:String(it.productId||''),
          productName:String(it.productName||'Produto'),
          ean:String(it.ean||''),
          systemStock:Math.max(0,Number(it.systemStock)||0),
          countedQty:Math.max(0,Number(it.countedQty)||0),
          sectionId:it.sectionId||null,
          unit:String(it.unit||'UN')
        })).filter(it=>it.productId)
      };
      localStorage.setItem(key,JSON.stringify(payload));
      setInventoryDraftStatus('Rascunho salvo automaticamente neste dispositivo às '+new Date().toLocaleTimeString('pt-BR',{hour:'2-digit',minute:'2-digit'})+'.');
    }catch(e){
      setInventoryDraftStatus('Não foi possível salvar o rascunho automaticamente. Verifique o espaço disponível no navegador.');
    }
  }
  function restoreInventoryDraft(user){
    const key=inventoryDraftKey(user);
    state.inventory=null;
    state.inventoryItems=[];
    if(!key){setInventoryDraftStatus('Entre no sistema para recuperar o rascunho.');return;}
    try{
      const raw=localStorage.getItem(key);
      if(!raw){setInventoryDraftStatus('Nenhum inventário salvo neste dispositivo.');renderInventory();return;}
      const draft=JSON.parse(raw);
      if(draft?.version!==1||!draft.inventory||!Array.isArray(draft.items))throw new Error('Rascunho inválido.');
      state.inventory={
        name:String(draft.inventory.name||'Inventário sem título'),
        sectionId:draft.inventory.sectionId||null,
        startedAt:draft.inventory.startedAt||new Date().toISOString()
      };
      state.inventoryItems=draft.items.filter(it=>it&&it.productId).map(it=>({
        productId:String(it.productId),
        productName:String(it.productName||'Produto'),
        ean:String(it.ean||''),
        systemStock:Math.max(0,Number(it.systemStock)||0),
        countedQty:Math.max(0,Number(it.countedQty)||0),
        sectionId:it.sectionId||null,
        unit:String(it.unit||'UN')
      }));
      $('inventoryName').value=state.inventory.name;
      setInventoryDraftStatus('Rascunho recuperado automaticamente deste dispositivo. Confira os itens antes de finalizar.');
      renderInventory();
    }catch(e){
      try{localStorage.removeItem(key);}catch(_){}
      state.inventory=null;state.inventoryItems=[];
      setInventoryDraftStatus('O rascunho salvo estava inválido e foi descartado. Você pode iniciar um novo inventário.');
      renderInventory();
    }
  }

      if(coverage.missing.length){
        const preview=coverage.missing.slice(0,5).map(p=>p.name||p.ean||'Produto').join(', ');
        const more=coverage.missing.length>5?' e mais '+(coverage.missing.length-5)+' produto(s)':'';
        const approved=window.confirm('ATENÇÃO: este inventário cobre '+coverage.counted+' de '+coverage.total+' produto(s). Ainda faltam '+coverage.missing.length+' produto(s) sem contagem.\n\nPendentes: '+preview+more+'.\n\nSe finalizar agora, somente os produtos contados terão seus saldos atualizados. Os produtos não contados NÃO serão zerados nem alterados.\n\nDeseja finalizar mesmo assim?');
        if(!approved)return;
      }
      const batch=db.batch(),invRef=db.collection('inventories').doc();
      batch.set(invRef,{name:state.inventory.name,sectionId:state.inventory.sectionId,startedAt:state.inventory.startedAt,finishedAt:firebase.firestore.FieldValue.serverTimestamp(),status:'finalizado',itemCount:state.inventoryItems.length,coverageTotal:coverage.total,coverageCounted:coverage.counted,coveragePercent:coverage.percent,userId:state.user.uid,userEmail:state.user.email||''});
      for(const it of state.inventoryItems){
        const pRef=db.collection('products').doc(it.productId);
        batch.update(pRef,{stock:Number(it.countedQty),updatedAt:firebase.firestore.FieldValue.serverTimestamp(),updatedBy:state.user.uid});
        const cRef=db.collection('counts').doc();
        batch.set(cRef,{...it,systemStock:Number(it.systemStock),countedQty:Number(it.countedQty),difference:Number(it.countedQty)-Number(it.systemStock),userId:state.user.uid,userEmail:state.user.email||'',createdAt:firebase.firestore.FieldValue.serverTimestamp(),inventoryId:invRef.id});
      }
      await batch.commit();state.inventory=null;state.inventoryItems=[];persistInventoryDraft();renderInventory();message('Inventário finalizado. Estoques atualizados e diferenças registradas no histórico; o rascunho local foi removido.','success');
    }catch(e){message('Não foi possível finalizar o inventário: '+e.message,'error');}}
