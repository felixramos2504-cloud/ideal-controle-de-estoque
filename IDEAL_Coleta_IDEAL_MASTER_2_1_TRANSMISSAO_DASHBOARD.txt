    for(let i=digits.length-1,weight=3;i>=0;i--,weight=weight===3?1:3)sum+=digits[i]*weight;

  }

      const freshByEan=new Map(products.filter(product=>product.ean).map(product=>[normalizeEan(product.ean),product]));
      for(let offset=0;offset<collectionItems.length;offset+=400){
        const batch=db.batch(),chunk=collectionItems.slice(offset,offset+400);
        chunk.forEach(item=>{
          const itemEan=normalizeEan(item.ean);
          const byEan=freshByEan.get(itemEan);
          const byId=freshProducts.get(item.id);
          const existing=byEan||(byId&&normalizeEan(byId.ean)===itemEan?byId:null);
          if(existing){
            batch.update(db.collection('products').doc(existing.id),{stock:Number(item.quantity),updatedAt:firebase.firestore.FieldValue.serverTimestamp(),updatedBy:auth.currentUser.uid});
          }else{
            const newId='ean_'+itemEan,occupied=freshProducts.get(newId);
            if(occupied&&normalizeEan(occupied.ean)!==itemEan)throw new Error('Conflito no identificador do EAN '+item.ean+'. Nenhuma linha desse lote foi enviada; avise a gerência.');
            const ref=db.collection('products').doc(newId);
            batch.set(ref,{ean:itemEan,name:'Produto EAN '+itemEan,sectionId:'',unit:'UN',stock:Number(item.quantity),createdAt:firebase.firestore.FieldValue.serverTimestamp(),createdBy:auth.currentUser.uid,updatedAt:firebase.firestore.FieldValue.serverTimestamp(),updatedBy:auth.currentUser.uid});
          }
        });
        await batch.commit();sent+=chunk.length;$('collectionStatus').textContent='Enviados '+fmt(sent)+' de '+fmt(collectionItems.length)+' produto(s)...';
      }
      try{
        await db.collection('auditLogs').add({action:'stock_collection_sent',entityType:'products',entityId:'collection',summary:'Coleta de estoque enviada: '+sent+' produto(s)',details:{rowsProcessed:sent,items:collectionItems.map(item=>({productName:item.name,ean:item.ean,newStock:Number(item.quantity)}))},userId:auth.currentUser.uid,userEmail:String(auth.currentUser.email||''),role:currentRole,createdAt:firebase.firestore.FieldValue.serverTimestamp()});
      }catch(auditError){console.warn('Não foi possível registrar a auditoria da coleta:',auditError);}
      collectionItems=[];selectedProduct=null;$('selectedProductCard').classList.add('hidden');$('eanInput').value='';
      saveCollectionDraft();renderCollection();const transmittedAt=new Date().toLocaleString('pt-BR');$('collectionStatus').textContent='Transmissão concluída às '+transmittedAt+'. Os saldos foram gravados no Firestore; o dashboard recebe as alterações automaticamente em tempo real.';
      showMessage(fmt(sent)+' produto(s) transmitidos. O dashboard principal receberá os novos saldos em tempo real.','success');
    }catch(err){
      $('collectionStatus').textContent='Falha no envio. Confira as permissões e tente novamente.';
      const denied=String(err.code||'').includes('permission-denied')||/permission|permissão/i.test(err.message||'');
      showMessage(denied?'O Firebase bloqueou a gravação. Publique no Console do Firebase as regras atualizadas do arquivo firestore.rules, que autorizam a inclusão automática de EANs coletados. Erro: '+(err.message||'permissão negada'):'Não foi possível enviar toda a coleta: '+(err.message||err),'error');
    }finally{busy=false;renderCollection();}
  }
  async function loadProfile(user){
    const snap=await db.collection('users').doc(user.uid).get();
    if(!snap.exists||snap.data().ativo===false||!allowedRoles.includes(snap.data().role)){await auth.signOut();throw new Error('Sua conta não possui um perfil ativo autorizado para coleta.');}
    currentRole=snap.data().role;currentUserId=user.uid;$('roleBadge').textContent='Perfil: '+currentRole;$('userEmail').textContent=user.email||'Conta autorizada';
    $('userBar').classList.remove('hidden');$('authPanel').classList.add('hidden');$('appPanel').classList.remove('hidden');
  }
  function attachDataListeners(){
    stopDataListeners();loadingProducts=true;$('sendBtn').disabled=true;
    dataUnsubs.push(db.collection('sections').orderBy('name').onSnapshot(snapshot=>{sections=snapshot.docs.map(doc=>({id:doc.id,...doc.data()}));renderCollection();},err=>showMessage('Não foi possível ler seções: '+err.message,'error')));
    dataUnsubs.push(db.collection('products').orderBy('name').onSnapshot(snapshot=>{
      products=snapshot.docs.map(doc=>({id:doc.id,...doc.data()}));loadingProducts=false;
      if(!draftRestored){restoreCollectionDraft();draftRestored=true;}
      collectionItems=collectionItems.map(item=>{const current=products.find(p=>p.id===item.id)||products.find(p=>normalizeEan(p.ean)===normalizeEan(item.ean));return current?{...item,id:current.id,isNew:false,name:current.name||item.name,ean:normalizeEan(current.ean||item.ean),sectionId:current.sectionId||item.sectionId,unit:current.unit||item.unit,currentStock:Number(current.stock||0)}:item;});
      renderCollection();saveCollectionDraft();
    },err=>{loadingProducts=false;renderCollection();showMessage('Não foi possível carregar o catálogo: '+err.message,'error');}));
  }

  });

  try{
    if(!window.IDEAL_FIREBASE_CONFIG||!window.IDEAL_FIREBASE_CONFIG.apiKey||String(window.IDEAL_FIREBASE_CONFIG.apiKey).includes('COLOQUE_'))throw new Error('firebase-config.js ainda não está configurado.');

  }catch(err){showMessage('Erro ao iniciar o Firebase: '+err.message,'error');}

  });
